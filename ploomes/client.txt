# ploomes/client.py
import requests
from typing import List, Tuple, Optional

# Wrapper de requests com retry e backoff para 429/5xx
from utils.http_retry import request_with_retry

class PloomesClient:
    """
    Cliente de integração com a API do Ploomes.
    Responsável por:
      - Listar negócios (Deals) com paginação
      - Ler OtherProperties com diferentes estratégias (fallbacks)
      - Atualizar campos (OtherProperties) de um negócio
    """
    def __init__(self, user_key: str):
        # URL base do ambiente do Ploomes
        self.base_url = "https://api2.ploomes.com"
        # Cabeçalhos padrão: User-Key + Content-Type
        self.headers = {"User-Key": user_key, "Content-Type": "application/json"}

    def iter_deals(self, filter_odata: str, top: int = 50):
        """
        Itera os negócios (Deals) que atendem ao filtro OData informado.

        - top: quantidade máxima de registros por página (default=50)
        - o método faz paginação usando $top e $skip até não haver mais registros.

        Uso típico:
            for deal in client.iter_deals(filter_odata, top=50):
                ...
        """
        skip = 0 # deslocamento da página atual
        while True:
            params = {
                "$filter": filter_odata,
                "$select": "Id,Title,StageId,PipelineId", # seleciona apenas campos básicos
                "$top": top,
                "$skip": skip
            }
            # GET com retry/backoff
            resp = request_with_retry("GET", f"{self.base_url}/Deals", headers=self.headers, params=params)
            data = resp.json()
            items = data.get("value", [])

            # Quando a API não retornar itens, chegamos ao fim da paginação
            if not items:
                break

            # Entrega cada negócio individualmente
            for item in items:
                yield item

            # Avança para a próxima "página"
            skip += top

    # ---------------- Estratégias internas para ler OtherProperties ----------------
    def _get_otherprops_via_deals_expand_by_filter(self, deal_id: int) -> list[dict]:
        """
        Estratégia 1: Usa /Deals com $filter=Id eq {id} e $expand=OtherProperties.
        Essa é a abordagem inspirada no teste que você confirmou que funciona
        (similar ao teste.py).
        """
        params = {
            "$top": 1,
            "$skip": 0,
            "$select": "*",
            "$expand": "OtherProperties",
            "$filter": f"Id eq {deal_id}",
            "$orderby": "Id desc",
        }
        resp = request_with_retry("GET", f"{self.base_url}/Deals", headers=self.headers, params=params)
        data = resp.json()
        values = data.get("value") or []
        if not values:
            return []
        # Pega o primeiro negócio retornado e devolve apenas a lista de OtherProperties
        return values[0].get("OtherProperties") or []

    # estratégia 2
    def _get_otherprops_via_deal_expand(self, deal_id: int) -> list[dict]:
        """
        Estratégia 2: Usa /Deals({id}) com $expand=OtherProperties.
        Traz o negócio diretamente pelo Id, expandindo apenas alguns campos de OtherProperties.
        """
        params = {"$expand": "OtherProperties($select=FieldKey,StringValue,DateTimeValue,ObjectValueName,EnumValueName,IntegerValue,ContactValueName)"}
        resp = request_with_retry("GET", f"{self.base_url}/Deals({deal_id})", headers=self.headers, params=params)

        # Em algumas respostas, o Ploomes retorna {"value": [ {...deal...} ]}
        value = (resp.json().get("value") or [None])[0]
        if not value:
            return []
        return value.get("OtherProperties") or []

    def _resolve_field_id(self, field_key: str) -> int:
        """
        Estratégia 3 (parte 1): Dado um FieldKey (ex.: 'deal_XXXX'),
        busca na entidade Fields para descobrir o FieldId numérico correspondente.
        """
        params = {"$filter": f"Key eq '{field_key}'", "$select": "Id,Key"}
        resp = request_with_retry("GET", f"{self.base_url}/Fields", headers=self.headers, params=params)
        items = resp.json().get("value", [])
        if not items or not isinstance(items[0].get("Id"), int):
            # Se não encontrar, gera erro explicando o problema
            raise RuntimeError(f"FieldKey '{field_key}' não encontrado em /Fields.")
        return items[0]["Id"]

    def _get_otherprops_by_fieldid(self, deal_id: int, field_id: int) -> list[dict]:
        """
        Estratégia 3 (parte 2): Usa o FieldId numérico para buscar OtherProperties
        específicos através de /Deals({id})/OtherProperties.
        """
        params = {
            "$filter": f"FieldId eq {field_id}",
            "$select": "FieldId,FieldKey,ContactValueName,ObjectValueName,EnumValueName,StringValue,DateTimeValue,IntegerValue",
            "$top": 50,
        }
        url = f"{self.base_url}/Deals({deal_id})/OtherProperties"
        resp = request_with_retry("GET", url, headers=self.headers, params=params)
        return resp.json().get("value", []) or []

    # ---------------- API pública para obter OtherProperties com fallbacks ----------------
    def get_otherprops_resilient(self, deal_id: int, prefer_field_key: str | None = None) -> list[dict]:
        """
        Tenta obter OtherProperties usando múltiplas estratégias, em ordem:

          1) /Deals?$filter=Id eq {id}&$expand=OtherProperties
          2) /Deals({id})?$expand=OtherProperties
          3) (opcional) /Fields -> FieldId -> /Deals({id})/OtherProperties?$filter=FieldId eq ...

        Se alguma der certo, retorna a lista de propriedades.
        Se todas falharem, retorna lista vazia.
        """

        # 1) Estratégia baseada no filtro por Id + expand
        try:
            props = self._get_otherprops_via_deals_expand_by_filter(deal_id)
            if props:
                return props
        except requests.HTTPError:
            # Se a API responder com HTTPError (ex.: 403/404), ignora e tenta próxima estratégia
            pass
        # 2) Estratégia baseada em /Deals({id}) direto
        try:
            props = self._get_otherprops_via_deal_expand(deal_id)
            if props:
                return props
        except requests.HTTPError:
            pass
        # 3) Estratégia baseada em FieldId (só faz sentido se soubermos o FieldKey preferencial)
        if prefer_field_key:
            try:
                fid = self._resolve_field_id(prefer_field_key)
                props = self._get_otherprops_by_fieldid(deal_id, fid)
                if props:
                    return props
            except requests.HTTPError:
                pass
        # Nenhuma estratégia funcionou
        return []

    # ---------------- Atualização de status no Ploomes ----------------
    def update_status(
            self,
            deal_id: int,
            field_key: str,
            descricao: str,
            date_field_key: Optional[str] = None,
            delivery_date: Optional[str] = None,
            additional_datetime_updates: Optional[List[Tuple[str, str]]] = None):

        """
        Atualiza o status do negócio no Ploomes via OtherProperties.

        Parâmetros:
          - deal_id: id do negócio
          - field_key: FieldKey do campo de descritivo de status (string)
          - descricao: texto do status (StringValue)
          - date_field_key: FieldKey da Previsão de Entrega (Data)
          - delivery_date: data/hora (ISO) da Previsão de Entrega
          - additional_datetime_updates: lista de (FieldKey, ISO datetime) para outros campos de data
            (ex.: Data de Entrega Atualizada)

        Regras:
          - Sempre atualiza o campo de descrição.
          - Previsão de Entrega (date_field_key) só é gravada se o campo ainda estiver vazio.
          - Campos adicionais de data são sempre gravados (sem checar se já existe valor).
        """

        payload = {
            "OtherProperties": [
                {
                    "FieldKey": field_key,
                    "StringValue": descricao or ""
                }
            ]
        }

        # Previsão de Entrega: só grava se ainda não houver valor no campo
        if date_field_key and delivery_date:
            campos = self.get_otherprops_resilient(deal_id, prefer_field_key=date_field_key)
            previsao_existente = next((c for c in campos if c.get("FieldKey") == date_field_key and c.get("DateTimeValue")), None)
            if not previsao_existente:
                payload["OtherProperties"].append({"FieldKey": date_field_key, "DateTimeValue": delivery_date})

        # Campos adicionais de data/hora (ex.: Data de Entrega Atualizada)
        for fk, dt_iso in (additional_datetime_updates or []):
            if fk and dt_iso:
                payload["OtherProperties"].append({"FieldKey": fk, "DateTimeValue": dt_iso})

        # PATCH com retry/backoff
        resp = request_with_retry("PATCH", f"{self.base_url}/Deals({deal_id})", headers=self.headers, json=payload)
        return resp.json()
